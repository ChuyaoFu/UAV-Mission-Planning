cmake_minimum_required(VERSION 3.10)
project(openvins_gftt_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs features2d highgui video)
set(OV_SOURCE "" CACHE PATH "Patched OpenVINS source")
set(EIGEN_SOURCE "" CACHE PATH "Eigen include root")
add_executable(gftt_native_test native_detector_test.cpp)
target_include_directories(gftt_native_test PRIVATE ${OpenCV_INCLUDE_DIRS} ${EIGEN_SOURCE} ${OV_SOURCE}/ov_core/src)
target_link_libraries(gftt_native_test ${OpenCV_LIBS})
if(ENABLE_SANITIZERS)
 target_compile_options(gftt_native_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(gftt_native_test PRIVATE -fsanitize=address,undefined)
endif()
if(ACTUAL_DETECTION_TEST)
 add_executable(actual_detection_test ${ACTUAL_DETECTION_TEST})
 target_include_directories(actual_detection_test PRIVATE ${OpenCV_INCLUDE_DIRS} ${EIGEN_SOURCE} ${OV_SOURCE}/ov_core/src)
 target_link_libraries(actual_detection_test ${OpenCV_LIBS})
 if(ENABLE_SANITIZERS)
  target_compile_options(actual_detection_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(actual_detection_test PRIVATE -fsanitize=address,undefined)
 endif()
endif()
if(OV_BASELINE_SOURCE)
 add_executable(baseline_fast_fingerprint baseline_fast_fingerprint.cpp)
 target_include_directories(baseline_fast_fingerprint PRIVATE ${OpenCV_INCLUDE_DIRS} ${EIGEN_SOURCE} ${OV_BASELINE_SOURCE}/ov_core/src)
 target_link_libraries(baseline_fast_fingerprint ${OpenCV_LIBS})
endif()
