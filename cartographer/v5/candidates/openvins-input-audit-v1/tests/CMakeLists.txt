cmake_minimum_required(VERSION 3.13)
project(readonly_sensor_entry_audit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
set(OV_SOURCE "" CACHE PATH "Pinned OpenVINS root")
set(EIGEN_SOURCE "" CACHE PATH "Eigen include root")
add_executable(native_logger_test native_logger_test.cpp)
target_include_directories(native_logger_test PRIVATE ${OpenCV_INCLUDE_DIRS} ${EIGEN_SOURCE} ${OV_SOURCE}/ov_core/src ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(native_logger_test ${OpenCV_LIBS})
if(ENABLE_SANITIZERS)
 target_compile_options(native_logger_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(native_logger_test PRIVATE -fsanitize=address,undefined)
endif()
