cmake_minimum_required(VERSION 3.13)
project(optional_retained_home_audit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(OV_SOURCE "" CACHE PATH "Patched OpenVINS source")
set(EIGEN_SOURCE "" CACHE PATH "Eigen include root")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/native_core_test.cpp
 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/build_native_core_test.py ${OV_SOURCE}/ov_msckf/src/state/StateHelper.cpp ${CMAKE_CURRENT_BINARY_DIR}/native_core_test.cpp
 DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/build_native_core_test.py ${OV_SOURCE}/ov_msckf/src/state/StateHelper.cpp)
add_executable(native_core_test ${CMAKE_CURRENT_BINARY_DIR}/native_core_test.cpp)
add_executable(native_pose_chart_test native_pose_chart_test.cpp)
add_executable(native_home_yaw_test native_home_yaw_test.cpp)
foreach(test native_core_test native_pose_chart_test native_home_yaw_test)
 target_include_directories(${test} PRIVATE ${EIGEN_SOURCE} ${OV_SOURCE}/ov_core/src ${CMAKE_CURRENT_SOURCE_DIR})
 if(ENABLE_SANITIZERS)
  target_compile_options(${test} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${test} PRIVATE -fsanitize=address,undefined)
 endif()
endforeach()
